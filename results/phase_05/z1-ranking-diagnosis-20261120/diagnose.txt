"""Replay the failed cap-one diagnostic with component accounting and abstention."""
import argparse
from bisect import bisect_right
from collections import Counter, defaultdict
import json
import re
from pathlib import Path
from statistics import median

from audit_battery_forecast import WINDOW_US, generation_times, predict, trace, validate, windows
from run_experiment import digest

out = Path(__file__).parent
prior = Path('results/phase_05/z1-closure-20261120/battery-forecast-v2.json')
saved = json.loads(prior.read_text())
table = {tuple(map(int, key.split('|'))): value for key, value in saved['calibration'].items()}
calibration_paths = [Path('results/phase_04') / name for name in
                     ('20261117T023634948551Z-B1-2', '20261117T023831400765Z-B2-2')]
model_file = Path('configs/power-z1-v1.json')
power = json.loads(model_file.read_text())
components = ('cpu', 'lpm', 'deep', 'tx', 'listen')
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--holdout', action='store_true')
args = parser.parse_args()
stationary = saved['validation']['stationary_static']['records']
evaluation_inputs = {}
if args.holdout:
    stationary = []
    cohort = []
    for mode in ('B1','B2'):
        run_id, = re.findall(r'^passed .*?/results/phase_04/(\S+)',
                             (out/f'holdout-{mode}.log').read_text(), re.M)
        path = Path('results/phase_04') / run_id
        manifest, _ = trace(path)
        config = manifest['config']
        if (manifest['mode'], manifest['effective_cap'], config['seed'], config['period_ms']) != (mode,1,123458,1000):
            raise ValueError('Wrong holdout configuration')
        cohort.append((manifest['pins'], manifest['clients'], manifest['platform'],
                       {k:v for k,v in manifest['source_hashes'].items() if k != 'firmware/generated-conf.h'},
                       config['power_model'],
                       tuple(config[k] for k in ('warmup_ms','readings_per_sensor','deadline_ms',
                                                 'drain_ms','energy_sample_ms','payload_budget'))))
        evaluation_inputs[str(path)] = digest(path/'manifest.json')
        stationary.extend(validate(table, path, False))
    if cohort[0] != cohort[1]:
        raise ValueError('Mismatched holdout cohort')

def component_costs(record, events):
    raw = [event for event in events if event['event_type'] == 'ENERGY_RAW']
    times = [event['sim_time_us'] for event in raw]
    start, end = record['start_us'], record['end_us']
    a, b = (raw[bisect_right(times, time)-1] for time in (start, end))
    denominator = 1000 * power['frequency']
    result = {}
    for key, current in zip(components, power['nanoamps']):
        coefficient = current * power['millivolts']
        cumulative_a = a[key] * coefficient // denominator
        cumulative_b = b[key] * coefficient // denominator
        result[key] = (cumulative_b - cumulative_a) * WINDOW_US / (end-start)
    if abs(sum(result.values())-record['observed_nj']) > 2:
        raise ValueError('Counter intervals do not reconcile to the forecast window')
    return result

# Estimate an empirical lower/upper forecast-error envelope from separate data.
# This is a one-cohort sensitivity check, not a statistical guarantee.
errors = defaultdict(list)
for path in calibration_paths:
    manifest, nodes = trace(path)
    if manifest['config']['seed'] != 223457:
        raise ValueError(path)
    for node, events in nodes.items():
        gen = generation_times(events)
        profile = manifest['config']['profile']
        p = predict(table, node, profile, 1, 100)
        errors[node, profile].extend(cost-p for _, _, cost in windows(events, gen[0], gen[-1]))

groups = defaultdict(dict)
traces = {}
for row in stationary:
    if row['cap'] != 1:
        continue
    directory = Path('results/phase_04') / row['run']
    if directory not in traces:
        _, traces[directory] = trace(directory)
    enriched = dict(row)
    enriched['components_nj'] = component_costs(row, traces[directory][row['node']])
    groups[row['node'],row['window']][row['profile']] = enriched

comparisons = []
for (node, window), rows in sorted(groups.items()):
    if set(rows) != {1,2}:
        continue
    midpoint = sum(r['predicted_nj'] for r in rows.values()) / 2
    observed = [p for p,r in rows.items() if r['observed_nj'] <= midpoint]
    baseline = [p for p,r in rows.items() if r['predicted_nj'] <= midpoint]
    intervals = {p: [r['predicted_nj']+min(0,*errors[node,p]),
                     r['predicted_nj']+max(0,*errors[node,p])] for p,r in rows.items()}
    yes = [p for p,(_,high) in intervals.items() if high <= midpoint]
    no = [p for p,(low,_) in intervals.items() if low > midpoint]
    comparisons.append(dict(node=node,window=window,budget_nj=midpoint,
        observed_affordable=observed,baseline_affordable=baseline,
        conservative_selected=yes[0] if len(yes)==1 and len(no)==1 else None,
        intervals=intervals,rows=rows))
outcomes = Counter('abstained' if c['conservative_selected'] is None else
                   'correct' if [c['conservative_selected']] == c['observed_affordable'] else 'wrong'
                   for c in comparisons)
abs_errors = [r['observed_nj']-r['predicted_nj'] for c in comparisons for r in c['rows'].values()]
forecast_gaps = [abs(c['rows'][1]['predicted_nj']-c['rows'][2]['predicted_nj']) for c in comparisons]
energy = [r['observed_nj'] for c in comparisons for r in c['rows'].values()]
listen_share = [r['components_nj']['listen']/r['observed_nj'] for c in comparisons for r in c['rows'].values()]
result = dict(evaluation_inputs=evaluation_inputs, forecast_file_sha256=digest(prior), power_model_sha256=digest(model_file),
    independent_error_envelope_cohort={str(p):digest(p/'manifest.json') for p in calibration_paths},
    matching_absolute_budget=sum(c['baseline_affordable']==c['observed_affordable'] for c in comparisons),
    aes_cheaper_observed=sum(c['rows'][1]['observed_nj'] < c['rows'][2]['observed_nj'] for c in comparisons),
    observed_neither_affordable=sum(not c['observed_affordable'] for c in comparisons),
    median_signed_forecast_underestimate_nj=median(abs_errors),
    median_forecast_action_gap_nj=median(forecast_gaps),
    median_observed_demand_nj=median(energy),
    median_listen_share=median(listen_share),
    conservative_outcomes=dict(outcomes), comparisons=comparisons)
(out/('holdout-result.json' if args.holdout else 'diagnosis.json')).write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='comparisons'}, indent=2))
