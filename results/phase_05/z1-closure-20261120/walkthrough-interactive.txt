"""Capture rendered dashboard pages with Firefox WebDriver."""
import base64
import json
import subprocess
import time
from pathlib import Path
from urllib.error import URLError
from urllib.request import Request, urlopen

out = Path(__file__).parent
driver_url = 'http://127.0.0.1:4445'

def request(path, payload=None, method=None):
    body = json.dumps(payload).encode() if payload is not None else None
    req = Request(driver_url + path, body, {'Content-Type': 'application/json'}, method=method)
    with urlopen(req, timeout=45) as response:
        return json.load(response)['value']

with (out / 'geckodriver-interactive.log').open('w') as log:
    driver = subprocess.Popen(['geckodriver', '--port', '4445'], stdout=log, stderr=log)
    try:
        for _ in range(20):
            try:
                request('/status')
                break
            except URLError:
                time.sleep(.25)
        session = request('/session', {'capabilities': {'alwaysMatch': {
            'browserName': 'firefox', 'pageLoadStrategy': 'eager',
            'moz:firefoxOptions': {'args': ['-headless']}}}})
        prefix = '/session/' + session['sessionId']
        try:
            request(prefix + '/window/rect', {'width': 1600, 'height': 1100})
            for name, path in [('latest', '/'), ('historical', '/historical'), ('campaign', '/campaign')]:
                request(prefix + '/url', {'url': 'http://127.0.0.1:5051' + path})
                for _ in range(30):
                    ready = request(prefix + '/execute/sync', {'script': '''
                        const page = arguments[0];
                        return page === 'latest' ||
                          (page === 'historical' && document.querySelectorAll('tbody tr').length > 2) ||
                          (page === 'campaign' && document.querySelectorAll('tbody tr').length > 7);
                        ''', 'args': [name]})
                    if ready:
                        break
                    time.sleep(1)
                if not ready:
                    raise RuntimeError(name + ' did not finish loading')
                page = request(prefix + '/execute/sync', {'script':
                    'return {title: document.title, excerpt: document.body.innerText.slice(0, 900), scrollWidth: document.body.scrollWidth}', 'args': []})
                print(name, json.dumps(page), flush=True)
                checks = request(prefix + '/execute/sync', {'script': '''
                    const page = arguments[0];
                    if (page === 'latest') {
                      const details = document.querySelector('details');
                      details.querySelector('summary').click();
                      if (!details.open || !details.querySelector('a[href]'))
                        throw new Error('Missing run evidence disclosure');
                      return {links: details.querySelectorAll('a').length};
                    }
                    if (page === 'historical') {
                      const select = document.querySelector('#filter-status');
                      const rows = document.querySelector('#hist-rows');
                      const before = rows.children.length;
                      select.value = 'failed';
                      select.dispatchEvent(new Event('change', {bubbles: true}));
                      if (select.value !== 'failed' || rows.children.length >= before ||
                          !rows.innerText.includes('failed'))
                        throw new Error('Archive status filter failed');
                      return {before, filtered: rows.children.length};
                    }
                    const select = document.querySelector('#group-select');
                    select.selectedIndex = 1;
                    select.dispatchEvent(new Event('change', {bubbles: true}));
                    if (!select.value || document.querySelector('#mode-rows').children.length < 1)
                      throw new Error('Campaign group filter failed');
                    document.querySelector('#runs-rows tr').click();
                    return {group: select.value};
                ''', 'args': [name]})
                if name == 'campaign':
                    for _ in range(20):
                        detail = request(prefix + '/execute/sync', {'script':
                            "return document.querySelector('#dp-body').innerText", 'args': []})
                        if detail and 'Loading…' not in detail:
                            break
                        time.sleep(1)
                    if not detail or 'Loading…' in detail or 'failed to load' in detail.lower():
                        raise RuntimeError('Campaign run detail did not load: ' + detail[:300])
                    checks['detail_text'] = detail[:180]
                print(name, json.dumps(checks), flush=True)
                png = request(prefix + '/screenshot')
                (out / (name + '-interactive.png')).write_bytes(base64.b64decode(png))
        finally:
            request(prefix, method='DELETE')
    finally:
        driver.terminate()
        driver.wait(timeout=5)
